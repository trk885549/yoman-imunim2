<script>

const SYSTEM_PROMPT = `You are an institutional-grade stock analyst. Return ONLY JSON following schema strictly.`;

function formatMoney(n){
  if(n==null) return "-";
  if(n>=1e12) return (n/1e12).toFixed(2)+"T";
  if(n>=1e9) return (n/1e9).toFixed(2)+"B";
  if(n>=1e6) return (n/1e6).toFixed(2)+"M";
  return n.toLocaleString();
}

function scoreColor(score){
  if(score>=5) return "var(--green)";
  if(score>=3) return "var(--amber)";
  return "var(--red)";
}

async function analyze(retry=false){
  const key = apiKey.value;
  const ticker = document.getElementById("ticker").value;
  if(!key || !ticker) return alert("Missing fields");

  btn.disabled=true;
  status.className="loading";
  status.innerText="Running analysis...";

  try{
    let res = await fetch("https://api.anthropic.com/v1/messages",{
      method:"POST",
      headers:{
        "Content-Type":"application/json",
        "x-api-key":key,
        "anthropic-version":"2023-06-01"
      },
      body:JSON.stringify({
        model:"claude-sonnet-4-20250514",
        max_tokens:4000,
        system:SYSTEM_PROMPT,
        tools:[{type:"web_search_20250305", name:"web_search"}],
        messages:[{role:"user", content:ticker}]
      })
    });

    let data = await res.json();
    let text = data.content.map(x=>x.text||"").join("");
    let match = text.match(/\{[\s\S]*\}/);

    if(!match) throw "Parse error";

    let d = JSON.parse(match[0]);
    render(d);

  }catch(e){
    if(!retry){
      return analyze(true);
    }
    app.innerHTML=`<div class="card red">Error: ${e}</div>`;
  }

  btn.disabled=false;
  status.innerText="";
}

function render(d){

let color = d.priceChange>=0?"green":"red";
let scoreWidth = (d.canslimScore/7)*100;

app.innerHTML=`

<div class="card">
<div><span class="live-dot"></span>${d.ticker} | ${d.companyName}</div>
<div class="price ${color}">$${d.currentPrice}</div>
<div class="${color}">
${d.priceChange} (${d.priceChangePct}%)
</div>
<div>Market Cap: ${d.marketCap}</div>
<div>52W: ${d.low52} - ${d.high52}</div>
</div>

<div class="card">
<h3>CAN SLIM Score</h3>
<div>${d.canslimScore}/7</div>
<div class="score-bar">
  <div class="score-fill" style="width:${scoreWidth}%; background:${scoreColor(d.canslimScore)}"></div>
</div>
</div>

<div class="grid">

<div class="card">
<h4>CAN SLIM Breakdown</h4>
${Object.entries(d.canslim).map(([k,v])=>
`<div>${k}: ${v.pass?"✓":"✗"} - ${v.value}</div>`
).join("")}
</div>

<div class="card">
<h4>Trend</h4>
<div>${d.trend.label}</div>
<div>${d.trend.detail}</div>
</div>

<div class="card">
<h4>Moving Averages</h4>
${Object.entries(d.movingAverages).map(([k,v])=>{
let dist=((d.currentPrice-v.price)/v.price*100).toFixed(1);
return `<div>${k}: ${v.price} (${dist}%) ${v.above?"↑":"↓"}</div>`;
}).join("")}
</div>

<div class="card">
<h4>Analysts</h4>
<div>${d.analysts.consensusLabel}</div>

<div class="bar"><div class="fill" style="width:${d.analysts.buy*10}%"></div></div>
Buy: ${d.analysts.buy}

<div class="bar"><div class="fill" style="width:${d.analysts.hold*10}%"></div></div>
Hold: ${d.analysts.hold}

<div class="bar"><div class="fill" style="width:${d.analysts.sell*10}%"></div></div>
Sell: ${d.analysts.sell}

<div>Target: $${d.analysts.targetPrice}</div>
</div>

<div class="card">
<h4>Fundamentals</h4>
${Object.entries(d.fundamentals).map(([k,v])=>`<div>${k}: ${v}</div>`).join("")}
</div>

<div class="card">
<h4>Patterns</h4>
${d.patterns.map(p=>
`<div class="pattern">
${p.icon} ${p.name} (${p.confidence})<br>
${p.desc}
</div>`
).join("")}
</div>

</div>

<div class="card">
<h4>AI Summary</h4>
${d.aiSummary}
</div>
`;
}
